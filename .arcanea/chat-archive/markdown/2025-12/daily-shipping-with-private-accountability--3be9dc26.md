CREATE OR REPLACE FUNCTION search_arcanea(
  query_text IN VARCHAR2,
  top_k IN NUMBER DEFAULT 10
) RETURN SYS_REFCURSOR IS
  result_cursor SYS_REFCURSOR;
  query_vector VECTOR(1536, FLOAT32);
BEGIN
  -- Generate embedding via OCI GenAI
  query_vector := DBMS_VECTOR.UTL_TO_EMBEDDING(
    query_text,
    JSON('{"provider":"ocigenai","model":"cohere.embed-english-v3.0"}')
  );
  
  OPEN result_cursor FOR
    SELECT id, title, content, 
           VECTOR_DISTANCE(content_vector, query_vector, COSINE) as similarity
    FROM arcanea_vectors
    ORDER BY similarity
    FETCH FIRST top_k ROWS ONLY;
    
  RETURN result_cursor;
