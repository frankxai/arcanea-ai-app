import React, { useState, useEffect } from 'react';
import { BarChart, Bar, XAxis, YAxis, Tooltip, ResponsiveContainer } from 'recharts';
import { Card, CardHeader, CardContent } from '@/components/ui/card';
import { Progress } from '@/components/ui/progress';
import { Check, X } from 'lucide-react';

const generateRandomData = () => {
  return [
    { name: 'Arcanea', progress: Math.floor(Math.random() * 100) },
    { name: 'Creation Resorts', progress: Math.floor(Math.random() * 100) },
    { name: 'Personal Brand', progress: Math.floor(Math.random() * 100) },
    { name: 'AI Education', progress: Math.floor(Math.random() * 100) },
  ];
};

const Dashboard = () => {
  const [projectData, setProjectData] = useState(generateRandomData());
  const [dailyTasks, setDailyTasks] = useState([
    { task: 'Morning Exercise', completed: false },
    { task: 'Meditate', completed: false },
    { task: 'Work on Arcanea', completed: false },
    { task: 'Call Family', completed: false },
  ]);

  useEffect(() => {
    const interval = setInterval(() => {
      setProjectData(generateRandomData());
    }, 5000);
    return () => clearInterval(interval);
  }, []);

  const toggleTask = (index) => {
    const newTasks = [...dailyTasks];
    newTasks[index].completed = !newTasks[index].completed;
    setDailyTasks(newTasks);
  };

  return (
    <div className="p-4 max-w-4xl mx-auto">
      <h1 className="text-2xl font-bold mb-4">Personal Growth Dashboard</h1>
      
      <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-4">
        <Card>
          <CardHeader>Project Progress</CardHeader>
          <CardContent>
            <ResponsiveContainer width="100%" height={200}>
              <BarChart data={projectData}>
                <XAxis dataKey="name" />
                <YAxis />
                <Tooltip />
                <Bar dataKey="progress" fill="#8884d8" />
              </BarChart>
            </ResponsiveContainer>
          </CardContent>
        </Card>

        <Card>
          <CardHeader>Daily Tasks</CardHeader>
          <CardContent>
            {dailyTasks.map((task, index) => (
              <div key={index} className="flex items-center mb-2">
                <button
                  onClick={() => toggleTask(index)}
                  className={`p-2 rounded-full mr-2 ${
                    task.completed ? 'bg-green-500' : 'bg-gray-300'
                  }`}
                >
                  {task.completed ? <Check size={16} /> : <X size={16} />}
                </button>
                <span className={task.completed ? 'line-through' : ''}>
                  {task.task}
                </span>
              </div>
            ))}
          </CardContent>
        </Card>
      </div>

      <Card>
        <CardHeader>Overall Progress</CardHeader>
        <CardContent>
          <Progress value={65} className="w-full" />
          <p className="text-center mt-2">65% towards your goals</p>
        </CardContent>
      </Card>
    </div>
  );
};

export default Dashboard;

import React, { useState, useEffect } from 'react';
import { Card, CardHeader, CardContent } from '@/components/ui/card';
import { Progress } from '@/components/ui/progress';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Check, X, Send } from 'lucide-react';

const ThesisAccountabilityAI = () => {
  const [progress, setProgress] = useState(0);
  const [dailyGoal, setDailyGoal] = useState('');
  const [aiMessage, setAiMessage] = useState('Hello! I\'m your Thesis Accountability AI. What\'s your goal for today?');
  const [userInput, setUserInput] = useState('');

  const updateProgress = (amount) => {
    setProgress(Math.min(100, Math.max(0, progress + amount)));
  };

  const handleSendMessage = () => {
    if (userInput.trim() === '') return;

    if (!dailyGoal) {
      setDailyGoal(userInput);
      setAiMessage(`Great! Your goal for today is: ${userInput}. Let's work on achieving that. How can I help you get started?`);
    } else {
      // Simple AI logic - in a real application, this would be more sophisticated
      if (userInput.toLowerCase().includes('done') || userInput.toLowerCase().includes('completed')) {
        setAiMessage('Excellent work! You\'re making great progress. What\'s your next step?');
        updateProgress(10);
      } else if (userInput.toLowerCase().includes('stuck') || userInput.toLowerCase().includes('help')) {
        setAiMessage('It\'s okay to face challenges. Let\'s break down your goal into smaller tasks. What\'s the first small step you can take?');
      } else {
        setAiMessage('Keep up the good work! Remember, consistent progress is key. What part of your thesis are you working on right now?');
      }
    }

    setUserInput('');
  };

  return (
    <Card className="w-full max-w-md mx-auto">
      <CardHeader>Thesis Accountability AI</CardHeader>
      <CardContent>
        <div className="mb-4">
          <div className="flex justify-between mb-1">
            <span>Overall Thesis Progress</span>
            <span>{progress}%</span>
          </div>
          <Progress value={progress} className="w-full" />
        </div>
        <div className="mb-4">
          <h3 className="font-semibold mb-2">Today's Goal:</h3>
          <p>{dailyGoal || 'Not set yet'}</p>
        </div>
        <div className="mb-4 h-40 overflow-y-auto bg-gray-100 p-2 rounded">
          <p className="mb-2"><strong>AI:</strong> {aiMessage}</p>
        </div>
        <div className="flex gap-2">
          <Input 
            value={userInput}
            onChange={(e) => setUserInput(e.target.value)}
            placeholder="Type your message..."
            onKeyPress={(e) => e.key === 'Enter' && handleSendMessage()}
          />
          <Button onClick={handleSendMessage}><Send size={16} /></Button>
        </div>
        <div className="flex justify-between mt-4">
          <Button onClick={() => updateProgress(-5)} variant="outline"><X size={16} className="mr-2" /> Setback</Button>
          <Button onClick={() => updateProgress(5)} variant="outline"><Check size={16} className="mr-2" /> Progress</Button>
        </div>
      </CardContent>
    </Card>
  );
};

export default ThesisAccountabilityAI;
