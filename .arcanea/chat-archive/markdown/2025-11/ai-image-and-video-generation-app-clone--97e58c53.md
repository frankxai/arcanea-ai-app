## 🛠️ VERCEL AI SDK APPROACH

### Why Vercel AI SDK vs Manual Integration

**Vercel AI SDK Benefits:**
- ✅ Unified interface for all AI models
- ✅ Built-in streaming support
- ✅ Automatic error handling & retries
- ✅ Edge runtime compatible
- ✅ TypeScript native
- ✅ Much less code (80% reduction)

**Perfect for:**
- Image generation (via Replicate, OpenAI)
- Text generation (Claude, GPT-4)
- Streaming responses
- Model switching

**NOT ideal for:**
- Video generation (Kling doesn't have Vercel SDK support yet)
- Custom queue management

### Hybrid Architecture

```
For Images:  Vercel AI SDK → Replicate/OpenAI
For Videos:  Direct API → Kling AI
For Prompts: Vercel AI SDK → Claude/GPT-4
```

-- Profiles table
CREATE TABLE profiles (

  email TEXT UNIQUE NOT NULL,

  avatar_url TEXT,
  credits INTEGER DEFAULT 10,
  tier TEXT DEFAULT 'free',
  created_at TIMESTAMPTZ DEFAULT NOW(),

);

-- Generations table
CREATE TABLE generations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  type TEXT NOT NULL CHECK (type IN ('image', 'video')),
  prompt TEXT NOT NULL,
  result_url TEXT NOT NULL,
  thumbnail_url TEXT,
  model_used TEXT NOT NULL,
  style_id UUID REFERENCES styles(id),
  is_public BOOLEAN DEFAULT true,
  likes_count INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Styles table
CREATE TABLE styles (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT UNIQUE NOT NULL,
  description TEXT,
  prompt_template TEXT NOT NULL,
  preview_url TEXT NOT NULL,
  category TEXT DEFAULT 'arcanea',
  frequency INTEGER,
  element TEXT
);

-- Likes table
CREATE TABLE likes (
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  generation_id UUID REFERENCES generations(id) ON DELETE CASCADE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  PRIMARY KEY (user_id, generation_id)
);

-- Indexes
CREATE INDEX idx_generations_user ON generations(user_id);
CREATE INDEX idx_generations_public ON generations(is_public) WHERE is_public = true;
CREATE INDEX idx_generations_created ON generations(created_at DESC);
CREATE INDEX idx_likes_generation ON likes(generation_id);

-- RLS Policies
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE generations ENABLE ROW LEVEL SECURITY;
ALTER TABLE likes ENABLE ROW LEVEL SECURITY;

-- Profiles: Users can read all, update own
CREATE POLICY "Profiles are viewable by everyone" ON profiles FOR SELECT USING (true);
CREATE POLICY "Users can update own profile" ON profiles FOR UPDATE USING (auth.uid() = id);

-- Generations: Public readable, own modifiable
CREATE POLICY "Public generations viewable" ON generations FOR SELECT USING (is_public = true OR user_id = auth.uid());
CREATE POLICY "Users can create generations" ON generations FOR INSERT WITH CHECK (user_id = auth.uid());
CREATE POLICY "Users can update own generations" ON generations FOR UPDATE USING (user_id = auth.uid());
CREATE POLICY "Users can delete own generations" ON generations FOR DELETE USING (user_id = auth.uid());

-- Likes: Users can like/unlike
CREATE POLICY "Anyone can view likes" ON likes FOR SELECT USING (true);
CREATE POLICY "Users can create likes" ON likes FOR INSERT WITH CHECK (user_id = auth.uid());
CREATE POLICY "Users can delete own likes" ON likes FOR DELETE USING (user_id = auth.uid());

-- Function to increment likes
CREATE OR REPLACE FUNCTION increment_likes(generation_uuid UUID)
RETURNS void AS $$
BEGIN
  UPDATE generations 
  SET likes_count = likes_count + 1 
  WHERE id = generation_uuid;
END;
$$ LANGUAGE plpgsql;

-- Function to decrement likes
CREATE OR REPLACE FUNCTION decrement_likes(generation_uuid UUID)
RETURNS void AS $$
BEGIN
  UPDATE generations 
  SET likes_count = likes_count - 1 
  WHERE id = generation_uuid;
END;
$$ LANGUAGE plpgsql;

-- Function to deduct credits
CREATE OR REPLACE FUNCTION deduct_credits(user_uuid UUID, amount INTEGER)
RETURNS void AS $$
BEGIN
  UPDATE profiles 
  SET credits = credits - amount 
  WHERE id = user_uuid AND credits >= amount;
END;
$$ LANGUAGE plpgsql;
